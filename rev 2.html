Funções

function add(x, y) {
  return x + y;
}

function somarTudo() {
  let total = 0;
  for (let i = 0; i < arguments.length; i++) {
    total += arguments[i];
  }
  return total;
}

Arrow Functions e Funções Anônimas

const somar = (a, b) => a + b;

const media = (...nums) => {
  let soma = nums.reduce((acc, val) => acc + val, 0);
  return soma / nums.length;
};

Classes e Construtores

  constructor(nome, email) {
    this.nome = nome;
    this.email = email;
  }
}

const joao = new Pessoa("João", "joao@email.com");
console.log(joao.nome);

Classes com Get e Set

class Produto {
  constructor(nome, preco) {
    this.nome = nome;
    this._preco = preco;
  }

  get preco() {
    return `R$ ${this._preco.toFixed(2)}`;
  }

  set preco(novoPreco) {
    if (novoPreco > 0) this._preco = novoPreco;
  }
}

Herança com Class

class Animal {
  constructor(nome) {
    this.nome = nome;
  }
  falar() {
    console.log(`${this.nome} faz um som.`);
  }
}

class Gato extends Animal {
  falar() {
    console.log(`${this.nome} diz: Miau!`);
  }
}
